find_package(KF6Runner REQUIRED)
add_library(meteo_runner_logic STATIC MeteoRunner.cpp MeteoRunner.hpp RunnerQuery.cpp RunnerQuery.hpp)
target_link_libraries(meteo_runner_logic PUBLIC meteo_ui KF6::Runner)
target_include_directories(meteo_runner_logic PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(meteo_runner MODULE Plugin.cpp)
target_link_libraries(meteo_runner PRIVATE meteo_runner_logic)
set_target_properties(meteo_runner PROPERTIES PREFIX "")
cmake_path(ABSOLUTE_PATH METEO_QML_INSTALL_DIR BASE_DIRECTORY "${CMAKE_INSTALL_PREFIX}" OUTPUT_VARIABLE meteo_ui_install)
file(RELATIVE_PATH meteo_ui_relative "${CMAKE_INSTALL_FULL_LIBDIR}/qt6/plugins/kf6/krunner" "${meteo_ui_install}")
set_target_properties(meteo_runner PROPERTIES INSTALL_RPATH "$ORIGIN/${meteo_ui_relative}")
install(TARGETS meteo_runner LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kf6/krunner")
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(test_runner "${PROJECT_SOURCE_DIR}/tests/test_runner.cpp")
    target_link_libraries(test_runner PRIVATE meteo_runner_logic Qt6::Test)
    krunner_configure_test(test_runner meteo_runner)
    add_test(NAME runner COMMAND test_runner)
    set_tests_properties(runner PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FORCE_STDERR_LOGGING=1" TIMEOUT 20)
endif()
