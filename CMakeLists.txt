cmake_minimum_required(VERSION 3.25)
project(meteo_pl VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)

option(METEO_BUILD_APP "Build the desktop application" ON)
option(METEO_BUILD_PLASMOID "Install the Plasma 6 widget" OFF)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Qml Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")

add_subdirectory(src)
if(METEO_BUILD_APP)
    find_package(KF6Kirigami REQUIRED)
    add_subdirectory(apps/meteo)
endif()
if(METEO_BUILD_PLASMOID)
    add_subdirectory(integrations/plasma)
endif()
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()
