cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" METEO_VERSION LIMIT_COUNT 1)
if(NOT METEO_VERSION MATCHES "^[0-9]+[.][0-9]+[.][0-9]+$")
    message(FATAL_ERROR "VERSION must contain MAJOR.MINOR.PATCH")
endif()
project(meteo_pl VERSION "${METEO_VERSION}" LANGUAGES CXX)
set(METEO_LICENSE "GPL-3.0-or-later")

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)
install(FILES integrations/plasma/package/contents/images/meteo.svg
    DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps" RENAME pl.meteo.desktop.svg)
install(FILES apps/meteo/pl.meteo.desktop.notifyrc DESTINATION "${CMAKE_INSTALL_DATADIR}/knotifications6")

option(METEO_BUILD_APP "Build the desktop application" ON)
set(METEO_PLASMOID_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(METEO_PLASMOID_DEFAULT ON)
endif()
option(METEO_BUILD_PLASMOID "Install the Plasma 6 widget" ${METEO_PLASMOID_DEFAULT})
option(METEO_BUILD_KRUNNER "Build the KRunner weather search plugin" ${METEO_BUILD_PLASMOID})
if(METEO_BUILD_PLASMOID AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The Plasma widget requires Linux; configure with METEO_BUILD_PLASMOID=OFF")
endif()

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Qml Quick QuickControls2)
find_package(Qt6 6.5 REQUIRED COMPONENTS Positioning)
find_package(KF6Notifications REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(Gettext REQUIRED)
if(WIN32 OR APPLE OR ANDROID)
    find_package(KF6IconThemes REQUIRED)
    find_package(ECM 6.0 REQUIRED NO_MODULE)
    list(APPEND CMAKE_MODULE_PATH "${ECM_MODULE_PATH}")
endif()
qt_standard_project_setup(REQUIRES 6.5)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")

add_subdirectory(src)
if(METEO_BUILD_APP)
    find_package(KF6Kirigami REQUIRED)
    add_subdirectory(apps/meteo)
endif()
if(METEO_BUILD_PLASMOID)
    add_subdirectory(integrations/plasma)
endif()
if(METEO_BUILD_KRUNNER)
    add_subdirectory(integrations/krunner)
endif()
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()

install(FILES README.md VERSION LICENSE DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/meteo-pl")
include(cmake/Packaging.cmake)
